Cardinal Finds — Entity-Relationship Diagram

Campus lost-and-found system · ITS122P Group 4 · created by database/schema.sql plus database/migrations/001_auth_security.sql (applied by database/seed.php)

erDiagram
    users {
        int      user_id                       PK
        varchar  first_name                    "not unique"
        varchar  last_name                     "not unique"
        varchar  email                         UK "login identifier"
        varchar  password_hash                 "bcrypt"
        enum     role                          "user | staff | admin | master_admin"
        tinyint  is_active
        tinyint  email_verified
        datetime email_verified_at
        char     email_verification_token_hash UK "SHA-256 of emailed token"
        datetime email_verification_expires_at
        tinyint  failed_login_attempts         "consecutive"
        datetime locked_at                     "null = not locked"
        datetime last_login_at
        datetime last_activity_at
        datetime created_at
        datetime updated_at
    }

    lost_reports {
        int      report_id       PK
        int      user_id         FK "reporter"
        varchar  item_name
        varchar  category
        text     description
        varchar  location_lost
        date     date_lost
        varchar  image_url
        enum     status          "open | matched | closed"
        int      matched_item_id FK "optional"
        datetime created_at
        datetime updated_at
    }

    found_items {
        int      item_id           PK
        int      user_id           FK "staff who logged it, or student who posted it"
        varchar  item_name
        varchar  category
        text     description
        text     private_details   "staff only"
        varchar  location_found
        varchar  storage_location
        date     date_found
        varchar  image_url
        enum     status            "stored | returned | disposed"
        datetime returned_at
        enum     moderation_status "pending | approved | rejected"
        int      moderated_by      FK "optional"
        text     moderation_note
        datetime moderated_at
        datetime created_at
        datetime updated_at
    }

    claims {
        int      claim_id          PK
        int      item_id           FK
        int      user_id           FK "claimant"
        int      report_id         FK "optional"
        text     proof_description
        enum     status            "pending | approved | rejected"
        date     date_claimed
        int      reviewed_by       FK "optional"
        text     review_note
        datetime reviewed_at
        datetime created_at
        datetime updated_at
    }

    user_sessions {
        int      session_id       PK
        int      user_id          FK
        char     token_hash       UK "SHA-256; raw token only in the browser session"
        enum     status           "active | logged_out | expired | revoked"
        varchar  ip_address
        varchar  user_agent
        datetime created_at
        datetime last_activity_at
        datetime expires_at
        datetime ended_at
    }

    security_events {
        bigint   event_id    PK
        int      user_id     FK "null for unknown accounts"
        varchar  email       "as typed"
        varchar  event_type  "login_success, login_failed, account_locked, ..."
        int      session_id
        varchar  ip_address
        varchar  user_agent
        json     metadata
        datetime created_at
    }
